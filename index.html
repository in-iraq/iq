// inIQ: التسجيل والتحقق من البريد وتسجيل الدخول وحفظ الحسابات في Google Sheet const SHEET_ID = '14hJwqQ0EMKf-sSkCBxH8FHBNaFkAic765WoN07FXMMA'; const CODE_TTL = 600; // مدة صلاحية رمز التأكيد بالثواني (10 دقائق) const MAX_ATTEMPTS = 5; // عدد محاولات إدخال رمز التأكيد const RESEND_WAIT = 60; // أقل مدة بين رسالتين لنفس البريد بالثواني const MAX_LOGIN_FAILS = 5; // عدد محاولات تسجيل الدخول الخاطئة قبل القفل const LOGIN_LOCK_TTL = 900; // مدة القفل بالثواني (15 دقيقة) function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return out({ ok: false, error: 'bad_request' }); } if (body.action === 'start') return out(start(body)); if (body.action === 'verify') return out(verify(body)); if (body.action === 'login') return out(login(body)); if (body.action === 'forgot') return out(forgot(body)); if (body.action === 'reset') return out(resetPassword(body)); return out({ ok: false, error: 'bad_request' }); } function out(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function sheet() { return SpreadsheetApp.openById(SHEET_ID).getSheets()[0]; } function normEmail(s) { return String(s || '').trim().toLowerCase(); } function emailExists(email) { const sh = sheet(); const last = sh.getLastRow(); if (last < 2) return false; const emails = sh.getRange(2, 4, last - 1, 1).getValues(); return emails.some(function (r) { return String(r[0]).trim().toLowerCase() === email; }); } // الخطوة 1: فحص البريد، وإن لم يكن مسجلًا يُرسل رمز التأكيد function start(b) { const name = String(b.name || '').trim().replace(/\s+/g, ' '); const gender = b.gender === 'male' ? 'ذكر' : b.gender === 'female' ? 'أنثى' : ''; const birth = String(b.birth || ''); const email = normEmail(b.email); const valid = name.split(' ').length >= 3 && gender && /^\d{4}-\d{2}-\d{2}$/.test(birth) && /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email); if (!valid) return { ok: false, error: 'invalid' }; // يوجد حساب مرتبط بالبريد if (emailExists(email)) return { ok: false, error: 'exists' }; const cache = CacheService.getScriptCache(); if (cache.get('sent:' + email)) return { ok: false, error: 'wait' }; const code = digits(6); cache.put('pending:' + email, JSON.stringify({ name: name, gender: gender, birth: birth, code: code, attempts: 0 }), CODE_TTL); cache.put('sent:' + email, '1', RESEND_WAIT); MailApp.sendEmail({ to: email, subject: 'رمز التأكيد - inIQ', body: 'رمز التأكيد الخاص بك: ' + code + '\nينتهي خلال 10 دقائق.' }); return { ok: true }; } // الخطوة 2: التحقق من الرمز ثم إنشاء الحساب وكلمة المرور function verify(b) { const email = normEmail(b.email); const cache = CacheService.getScriptCache(); const raw = cache.get('pending:' + email); if (!raw) return { ok: false, error: 'expired' }; const p = JSON.parse(raw); if (p.attempts >= MAX_ATTEMPTS) { cache.remove('pending:' + email); return { ok: false, error: 'too_many' }; } if (String(b.code || '').trim() !== p.code) { p.attempts += 1; cache.put('pending:' + email, JSON.stringify(p), CODE_TTL); return { ok: false, error: 'wrong_code' }; } const lock = LockService.getScriptLock(); lock.waitLock(20000); try { if (emailExists(email)) return { ok: false, error: 'exists' }; const password = 'iniq:' + digits(8); const salt = Utilities.getUuid(); const stored = salt + '$' + sha256(salt + password); // يُحفظ الهاش فقط const when = Utilities.formatDate(new Date(), 'Asia/Baghdad', 'yyyy-MM-dd HH:mm'); const sh = sheet(); const row = sh.getLastRow() + 1; sh.getRange(row, 1, 1, 6) .setNumberFormat('@') .setValues([[p.name, p.gender, p.birth, email, stored, when]]); cache.remove('pending:' + email); return { ok: true, password: password }; } finally { lock.releaseLock(); } } // تسجيل الدخول: يتحقق من البريد وكلمة المرور ويرجع الاسم function login(b) { const email = normEmail(b.email); const password = String(b.password || '').trim(); if (!email || !password) return { ok: false, error: 'invalid_login' }; const cache = CacheService.getScriptCache(); const failKey = 'fail:' + email; const fails = Number(cache.get(failKey) || 0); if (fails >= MAX_LOGIN_FAILS) return { ok: false, error: 'locked' }; const sh = sheet(); const last = sh.getLastRow(); if (last >= 2) { const rows = sh.getRange(2, 1, last - 1, 6).getValues(); for (let i = 0; i < rows.length; i++) { if (String(rows[i][3]).trim().toLowerCase() === email) { const parts = String(rows[i][4]).split('$'); if (parts.length === 2 && sha256(parts[0] + password) === parts[1]) { cache.remove(failKey); return { ok: true, name: String(rows[i][0]) }; } break; } } } cache.put(failKey, String(fails + 1), LOGIN_LOCK_TTL); return { ok: false, error: 'invalid_login' }; // نفس الرسالة للبريد غير المسجل وللكلمة الخاطئة } // رقم صف البريد في الشيت (0 إن لم يوجد) function findRow(email) { const sh = sheet(); const last = sh.getLastRow(); if (last < 2) return 0; const emails = sh.getRange(2, 4, last - 1, 1).getValues(); for (let i = 0; i < emails.length; i++) { if (String(emails[i][0]).trim().toLowerCase() === email) return i + 2; } return 0; } // نسيت كلمة المرور، الخطوة 1: يرسل رمز التأكيد إلى البريد المسجل function forgot(b) { const email = normEmail(b.email); if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) return { ok: false, error: 'invalid' }; if (!findRow(email)) return { ok: false, error: 'not_found' }; const cache = CacheService.getScriptCache(); if (cache.get('rsent:' + email)) return { ok: false, error: 'wait' }; const code = digits(6); cache.put('reset:' + email, JSON.stringify({ code: code, attempts: 0 }), CODE_TTL); cache.put('rsent:' + email, '1', RESEND_WAIT); MailApp.sendEmail({ to: email, subject: 'استعادة كلمة المرور - inIQ', body: 'رمز استعادة كلمة المرور: ' + code + '\nينتهي خلال 10 دقائق.\nإذا لم تطلب ذلك فتجاهل هذه الرسالة.' }); return { ok: true }; } // نسيت كلمة المرور، الخطوة 2: يتحقق من الرمز ويولّد كلمة مرور جديدة function resetPassword(b) { const email = normEmail(b.email); const cache = CacheService.getScriptCache(); const raw = cache.get('reset:' + email); if (!raw) return { ok: false, error: 'expired' }; const p = JSON.parse(raw); if (p.attempts >= MAX_ATTEMPTS) { cache.remove('reset:' + email); return { ok: false, error: 'too_many' }; } if (String(b.code || '').trim() !== p.code) { p.attempts += 1; cache.put('reset:' + email, JSON.stringify(p), CODE_TTL); return { ok: false, error: 'wrong_code' }; } const lock = LockService.getScriptLock(); lock.waitLock(20000); try { const row = findRow(email); if (!row) return { ok: false, error: 'not_found' }; const password = 'iniq:' + digits(8); const salt = Utilities.getUuid(); sheet().getRange(row, 5) .setNumberFormat('@') .setValue(salt + '$' + sha256(salt + password)); // تُستبدل البصمة القديمة cache.remove('reset:' + email); cache.remove('fail:' + email); return { ok: true, password: password }; } finally { lock.releaseLock(); } } // أرقام عشوائية من معرّف UUID function digits(n) { let result = ''; while (result.length < n) { const hex = Utilities.getUuid().replace(/-/g, ''); for (let i = 0; i < hex.length - 1 && result.length < n; i += 2) { const v = parseInt(hex.substr(i, 2), 16); if (v < 250) result += String(v % 10); } } return result; } function sha256(text) { return Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, text, Utilities.Charset.UTF_8 ).map(function (b) { return ('0' + (b & 0xff).toString(16)).slice(-2); }).join(''); }
